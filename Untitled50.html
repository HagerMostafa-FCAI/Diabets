# Function that caps outliers using the IQR (Interquartile Range) method
def cap_outliers_iqr(series, factor=1.5):
    # Calculate the first and third quartiles
    q1, q3 = series.quantile(0.25), series.quantile(0.75)
    # Interquartile range = Q3 - Q1
    iqr = q3 - q1
    # Define the lower and upper bounds for acceptable values
    lower, upper = q1 - factor * iqr, q3 + factor * iqr
    # Clip values outside the bounds to the nearest bound
    return series.clip(lower, upper)

# All feature columns (everything except the target column Outcome)

# Apply outlier capping to each feature column
for col in numeric_cols:
    df_clean[col] = cap_outliers_iqr(df_clean[col])

# Create a copy of the cleaned data for feature engineering
df_fe = df_clean.copy()

# Group age into ranges (bins)
df_fe["AgeGroup"] = pd.cut(df_fe["Age"], bins=[20, 30, 40, 50, 60, 100],
                            labels=["21-30", "31-40", "41-50", "51-60", "60+"])

# Categorize BMI into standard weight classes
df_fe["BMI_Category"] = pd.cut(df_fe["BMI"], bins=[0, 18.5, 25, 30, 100],
                                labels=["Underweight", "Normal", "Overweight", "Obese"])

# Categorize glucose levels: normal, prediabetic, diabetic
df_fe["Glucose_Category"] = pd.cut(df_fe["Glucose"], bins=[0, 99, 125, 300],
                                    labels=["Normal", "Prediabetic", "Diabetic"])

# Ratio of insulin to glucose
df_fe["Insulin_Glucose_Ratio"] = df_fe["Insulin"] / df_fe["Glucose"]
# Interaction feature: BMI multiplied by the diabetes pedigree function
df_fe["BMI_Pedigree_Interaction"] = df_fe["BMI"] * df_fe["DiabetesPedigreeFunction"]

# Count how many risk indicators are abnormal for each row
def count_abnormal(row):
    # Start with zero flags
    flags = 0
    # Each condition that is True adds 1: high glucose, high blood pressure, high BMI, high skin thickness
    flags += row["Glucose"] > 125
    flags += row["BloodPressure"] > 90
    flags += row["BMI"] > 30
    flags += row["SkinThickness"] > 40
    return flags

# Apply the function row by row to create the abnormal-count feature
df_fe["Abnormal_Count"] = df_fe.apply(count_abnormal, axis=1)
